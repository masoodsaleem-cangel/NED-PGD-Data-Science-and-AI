cmake_minimum_required(VERSION 3.20)
project(TokenScope VERSION 1.0 LANGUAGES C CXX)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "Select a Windows x64 compiler kit.")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED YES)
set(CMAKE_CXX_EXTENSIONS NO)
option(TOKENSCOPE_CAPTURE "Build the local GGUF capture worker" ON)
option(TOKENSCOPE_VIEWER "Build the OpenGL viewer" ON)
set(LLAMA_ROOT "$ENV{LLAMA_ROOT}" CACHE PATH "Matching llama.cpp SDK or b10816 Windows runtime folder")
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
function(portable_exe name)
    if(MINGW)
        target_link_options(${name} PRIVATE -static -static-libgcc -static-libstdc++)
    elseif(MSVC)
        target_compile_options(${name} PRIVATE /utf-8 /bigobj)
    endif()
    target_include_directories(${name} PRIVATE vendor/json src)
    target_link_libraries(${name} PRIVATE shell32)
endfunction()

if(TOKENSCOPE_CAPTURE)
    include(cmake/LlamaSDK.cmake)
    add_executable(Capture src/capture.cpp)
    portable_exe(Capture)
    target_include_directories(Capture PRIVATE ${LLAMA_INCLUDE})
    target_link_libraries(Capture PRIVATE ${LLAMA_LIBRARIES})
    add_custom_command(TARGET Capture POST_BUILD
        COMMAND ${CMAKE_COMMAND} -E copy_if_different ${LLAMA_DLLS} "$<TARGET_FILE_DIR:Capture>"
        COMMAND_EXPAND_LISTS VERBATIM)
endif()

if(TOKENSCOPE_VIEWER)
    set(g vendor/glfw)
    file(GLOB glfw_sources CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/${g}/src/*.c")
    add_library(glfw STATIC ${glfw_sources})
    target_include_directories(glfw PUBLIC ${g}/include)
    target_compile_definitions(glfw PRIVATE _GLFW_WIN32 UNICODE _UNICODE WINVER=0x0601 _WIN32_WINNT=0x0601)
    target_link_libraries(glfw PUBLIC gdi32 user32 shell32)
    add_library(glad STATIC vendor/glad/glad.c)
    target_include_directories(glad PUBLIC vendor/glad/include)
    set(i vendor/imgui)
    add_library(imgui STATIC ${i}/imgui.cpp ${i}/imgui_draw.cpp ${i}/imgui_tables.cpp
        ${i}/imgui_widgets.cpp ${i}/backends/imgui_impl_glfw.cpp ${i}/backends/imgui_impl_opengl3.cpp)
    target_include_directories(imgui PUBLIC ${i} ${i}/backends)
    target_link_libraries(imgui PRIVATE glfw opengl32)
    add_executable(TokenScope WIN32 src/viewer.cpp src/renderer.cpp src/session.cpp src/answer.cpp)
    portable_exe(TokenScope)
    target_link_libraries(TokenScope PRIVATE glfw glad imgui opengl32 comdlg32 gdi32)
    add_custom_command(TARGET TokenScope POST_BUILD COMMAND ${CMAKE_COMMAND} -E
        copy_directory "${CMAKE_CURRENT_SOURCE_DIR}/examples" "$<TARGET_FILE_DIR:TokenScope>/examples")
endif()
